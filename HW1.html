  1. Write a function nextHappy() that given a positive integer, computes the next value in the happy-number sequence. Be sure to provide a reasonable error on an invalid input. Be sure to document your function (see instructions above).
nextHappy <- function(input) {
  
  digits <- as.numeric(strsplit(as.character(input), "")[[1]])
  number <- 0
  for (digit in digits) {
    number <- number + (digit * digit)
  }
  return(number)
}

  1. Create a function happySequence() that returns the sequence beginning at a given number. Use your nextHappy() function to perform the calculation. Be sure to provide a reasonable error on an invalid input. Be sure to document your function (see instructions above).
happySequence <- function(input) {
  seen <- c()
  curr <- input
  sequence <- c(input)
  
  while(curr != 1 && !(curr %in% seen)) {
    seen <- c(seen, curr)
    curr <- nextHappy(curr)
    sequence <- c(sequence, curr)
    
  }
  
  is_happy <- (curr == 1)
  
  if (is_happy) {
    return(list(
      sequence = sequence,
      isHappy = is_happy
      ))
  } else {
    return(list(
      sequence = c(),
      isHappy = is_happy
      ))
  }

}
